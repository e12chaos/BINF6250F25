# --- 1. Definition of Key Algorithm Variables ---
#
#   --- E-Step (Expectation) Variables ---
#   alpha (table, T x N):
#       - The log-forward probabilities.
#       - alpha[t, i] = log(P(obs[1]...obs[t], state[t] = i))
#
#   beta (table, T x N):
#       - The log-backward probabilities.
#       - beta[t, i] = log(P(obs[t+1]...obs[T] | state[t] = i))
#
#   gamma (table, T x N):
#       - The posterior (posterior) probabilities.
#       - gamma[t, i] = P(state[t] = i | all_obs)
#       - This is the "Expected State Count" for a single position.
#
#   log_xi_ij (scalar, calculated at each step t, i, j):
#       - The joint log-posterior for a transition.
#       - log(P(state[t]=i, state[t+1]=j | all_obs))
#       - This is the "Expected Transition Count" for a single transition.
#
#   --- M-Step (Maximization) Variables ---
#   e_init_counts (dict, size N):
#       - Expected counts for starting in each state.
#
#   e_trans_numer (dict, size N x N):
#       - Numerator. Expected counts of transitioning from state 'i' to state 'j'.
#
#   e_trans_denom (dict, size N):
#       - Denominator. Expected counts of transitioning from state 'i' to *any* state.
#
#   e_emit_numer (dict, size N x M):
#       - Numerator. Expected counts of emitting observation 'k' from state 'i'.
#
#   e_emit_denom (dict, size N):
#       - Denominator. Expected counts of emitting *any* observation from state 'i'.
#
#   pseudocount (scalar):
#       - A small value added to all 'e_' counts to prevent zero probabilities.




# --- 2. The HMM Model Class ---
# An object-oriented structure to hold parameters and training logic.

class HMM_Model:
    
    # --- Initialization ---
    function __init__(self, init_probs_dict, trans_probs_dict, emit_probs_dict, pseudocount=0.01):
        # Stores the initial parameter guesses (dictionaries)
        self.init_probs = init_probs_dict
        self.trans_probs = trans_probs_dict
        self.emit_probs = emit_probs_dict
        
        # Stores the small value for smoothing
        self.pseudocount = pseudocount
        
        # Infers the set of states from the initial probabilities
        self.states = list(init_probs_dict.keys())
        
        # Infers the observation alphabet from the first state's emissions
        first_state = self.states[0]
        self.obs_alphabet = list(emit_probs_dict[first_state].keys())
        
        # Creates the observation-to-index map required by the algorithms
        self.obs_map = {symbol: i for i, symbol in enumerate(self.obs_alphabet)}

    # --- Training Loop (Baum-Welch) ---
    function baum_welch_train(self, training_obs_list, 
                             max_iterations=100, 
                             convergence_threshold=1e-5,
                             check_frequency=10):
        
        # Stores the log-likelihood from the previous convergence check
        previous_check_log_likelihood = -infinity

        for iter from 0 to max_iterations:
            
            # --- 2. Expectation Step (e-step) ---
            
            # Accumulates expected counts from all sequences for this iteration.
            # Initializes with a pseudocount to prevent zero probabilities.
            e_init_counts = {state: self.pseudocount for state in self.states}
            
            e_trans_numer = {s_i: {s_j: self.pseudocount for s_j in self.states} for s_i in self.states}
            e_trans_denom = {s_i: (self.pseudocount * len(self.states)) for s_i in self.states}
            
            e_emit_numer = {s_i: {obs: self.pseudocount for obs in self.obs_alphabet} for s_i in self.states}
            e_emit_denom = {s_i: (self.pseudocount * len(self.obs_alphabet)) for s_i in self.states}
            
            # Stores the total log-likelihood for this current iteration
            current_iter_log_likelihood = 0

            # Converts current parameter dictionaries to matrices once per iteration
            init_matrix, trans_matrix, emit_matrix = call convert_hmm_params_to_matrices(
                self.states, self.obs_map, self.init_probs, self.trans_probs, self.emit_probs
            )
            
            # Converts matrices to log-space for stable calculation
            with np.errstate(divide='ignore'):
                log_init = np.log(init_matrix)
                log_trans = np.log(trans_matrix)
                log_emit = np.log(emit_matrix)

            # Iterates through each observation sequence
            for seq_str in training_obs_list:
                
                obs_seq = [self.obs_map[obs] for obs in seq_str]
                T = len(obs_seq)

                # Runs forward-backward using the *current* model parameters
                # This calculates the posterior probabilities needed for the e-step
                
                # Calls the 'forward' function (your version)
                alpha_table, seq_log_prob = call forward(
                    obs_seq, self.states, init_matrix, trans_matrix, emit_matrix
                )
                
                # Calls the 'backward' function
                beta_table = call backward(
                    obs_seq, self.states, trans_matrix, emit_matrix
                )

                # --- Combine to get Gamma and Xi ---
                # gamma = P(state[t]=i | all_obs)
                log_gamma_unnormalized = alpha_table + beta_table
                log_posterior_gamma = log_gamma_unnormalized - seq_log_prob
                gamma = np.exp(log_posterior_gamma) # Convert to normal probability space
                
                current_iter_log_likelihood += seq_log_prob
                
                # --- Accumulate Expected Counts ---
                
                # 1. Accumulates expected initial state counts (from gamma[0])
                for i, state_i in enumerate(self.states):
                    e_init_counts[state_i] += gamma[0, i]

                # 2. Accumulates expected transition counts (from Xi)
                for t from 0 to T-2: # Loop up to the second-to-last time step
                    obs_t_plus_1 = obs_seq[t+1]
                    
                    for i, state_i in enumerate(self.states):
                        e_trans_denom[state_i] += gamma[t, i] # Denominator for A[i,j]
                        
                        for j, state_j in enumerate(self.states):
                            # Calculates log(xi[t, i, j]) using pre-computed values
                            # log_xi = log(alpha[t,i] * A[i,j] * B[j, obs[t+1]] * beta[t+1, j] / P(O))
                            log_xi_ij = (alpha_table[t, i] + 
                                         log_trans[i, j] + 
                                         log_emit[j, obs_t_plus_1] + 
                                         beta_table[t + 1, j] - 
                                         seq_log_prob)
                            
                            # Adds the probability (not log-prob) to the numerator
                            e_trans_numer[state_i][state_j] += np.exp(log_xi_ij)
                
                # 3. Accumulates expected emission counts (from Gamma)
                for t from 0 to T-1: # Loop over all time steps
                    current_obs_symbol = seq_str[t]
                    for i, state_i in enumerate(self.states):
                        e_emit_denom[state_i] += gamma[t, i] # Denominator for B[i,k]
                        e_emit_numer[state_i][current_obs_symbol] += gamma[t, i]

            # --- 3. Maximization Step (m-step) ---
            # Re-estimates model parameters by normalizing the expected counts
            
            # 1. Updates initial probabilities
            total_init = sum(e_init_counts.values())
            for state_i in self.states:
                self.init_probs[state_i] = e_init_counts[state_i] / total_init

            # 2. Updates transition probabilities
            for state_i in self.states:
                for state_j in self.states:
                    self.trans_probs[state_i][state_j] = e_trans_numer[state_i][state_j] / e_trans_denom[state_i]
            
            # 3. Updates emission probabilities
            for state_i in self.states:
                for obs in self.obs_alphabet:
                    self.emit_probs[state_i][obs] = e_emit_numer[state_i][obs] / e_emit_denom[state_i]

            # --- 4. Checks for Convergence (Efficiently) ---
            # Checks convergence only at specified intervals to save computation
            if iter > 0 and (iter % check_frequency == 0):
                
                likelihood_change = abs(current_iter_log_likelihood - previous_check_log_likelihood)
                
                if likelihood_change < convergence_threshold:
                    print "Convergence reached at iteration {iter}."
                    break loop # Terminate the loop early
                
                # Updates the likelihood for the next check
                previous_check_log_likelihood = current_iter_log_likelihood
        
        # Loop finishes
        return self.init_probs, self.trans_probs, self.emit_probs, current_iter_log_likelihood


# --- 3. Meta-Algorithm for Finding Global Optimum ---
# This function addresses the local optima problem by using multiple random restarts.

function find_best_hmm(training_obs_list, num_restarts=10, max_iterations_per_run=50, pseudocount=0.01):
    
    best_overall_model_params = null
    best_overall_log_likelihood = -infinity

    print "Starting {num_restarts} random restarts to find global optimum..."

    for i from 1 to num_restarts:
        print "--- Restart {i} ---"
        
        # 1. Randomizes initial parameters for this run
        init_guesses, trans_guesses, emit_guesses = call create_random_initial_params()
        
        # 2. Creates a new model instance with these random guesses
        temp_model = HMM_Model(init_guesses, trans_guesses, emit_guesses, pseudocount)
        
        # 3. Runs the baum_welch algorithm to find *one* local optimum
        (local_pi, local_A, local_B, local_log_likelihood) = temp_model.baum_welch_train(
            training_obs_list, 
            max_iterations=max_iterations_per_run
        )
        
        print "Restart {i} finished with Log-Likelihood: {local_log_likelihood}"

        # 4. Compares and stores the best log-likelihood found so far
        if local_log_likelihood > best_overall_log_likelihood:
            best_overall_log_likelihood = local_log_likelihood
            best_overall_model_params = (local_pi, local_A, local_B)
            
    # After all restarts, returns the parameters from the best-scoring run
    return best_overall_model_params, best_overall_log_likelihood

function generate_normalized_random_vector(size):
'''
Generates a vector of random numbers that sum to 1.0.
Args:
    size (int): The number of elements in the vector.

Returns:
    A list or array of floats, length 'size', summing to 1.0.
'''

vector = create_empty_array(size)
total_sum = 0

# 1. Generate 'size' random floats
for i from 0 to size-1:
    # Generates a random float, e.g., between 0.0 and 1.0
    val = random_float(0.0, 1.0)
    vector[i] = val
    total_sum = total_sum + val

# 2. Handle the (very unlikely) edge case where all random numbers are 0
if total_sum == 0:
    # Assign a uniform probability to all elements
    for i from 0 to size-1:
        vector[i] = 1.0 / size
else:
    # 3. Normalize the vector
    # Divides each element by the total sum
    for i from 0 to size-1:
        vector[i] = vector[i] / total_sum
        
return vector
